# Day 04 — Network Exposure and Firewall Review
# Run only on systems you own or are authorized to assess.
# Commands inspect local network state and do not alter configuration.

## 1. Local network baseline
ip -brief address
ip route

## 2. Listening services
ss -tulpn
sudo ss -tulpn

## 3. Firewall and packet-filtering state
sudo ufw status verbose
sudo nft list ruleset

## 4. Service exposure review
systemctl list-units --type=service --state=running --no-pager
systemctl is-enabled ssh
systemctl is-active ssh
systemctl is-enabled avahi-daemon
systemctl is-active avahi-daemon
